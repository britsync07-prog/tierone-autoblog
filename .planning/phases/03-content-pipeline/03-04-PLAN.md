lib/pipeline/draft-schema.ts (new — single source for ALL post frontmatter):
```typescript
import { z } from 'zod';
import { SILO_SLUGS } from '../silos';

export const comparisonColumnSchema = z.object({
  id: z.string().min(1),
  label: z.string().min(1),
});
export const comparisonRowSchema = z.object({
  cells: z.array(z.string().min(1)), // row width vs columns enforced by comparisonSchema .refine below
  best: z.boolean().optional(),
});
export const comparisonSchema = z
  .object({
    title: z.string().min(1).optional(),
    columns: z.array(comparisonColumnSchema).min(2).max(6),
    rows: z.array(comparisonRowSchema).min(2).max(12),
    note: z.string().optional(),
  })
  .refine((c) => c.rows.every((r) => r.cells.length === c.columns.length), {
    message: 'every comparison row must have one cell per column',
  });

/** Image slots (Plan 03-05 populates; optional for seeds and drafts until 03-05 runs). */
export const heroImageSchema = z.object({
  src: z.string().regex(/^\/images\/[a-z0-9]+(?:-[a-z0-9]+)*\/[a-z0-9]+(?:-[a-z0-9]+)*\.(?:avif|webp)$/),
  alt: z.string().min(10),
});

/** The full post frontmatter contract. Existing fields byte-compat with the
 * Phase-1 collection schema (title 50-60, description 150-160, slug regex,
 * ISO datetimes, faq 3-5, silo enum, tldr <=100 words). */
export const draftSchema = z.object({
  title: z.string().min(50).max(60),
  description: z.string().min(150).max(160),
  slug: z.string().regex(/^[a-z0-9]+(?:-[a-z0-9]+)*$/),
  publishedAt: z.string().datetime(),
  updatedAt: z.string().datetime(),
  faq: z.array(z.object({ q: z.string().min(1), a: z.string().min(1) })).min(3).max(5),
  silo: z.enum(SILO_SLUGS),
  tldr: z
    .string()
    .min(1)
    .refine((v) => v.split(/\s+/).filter(Boolean).length <= 100, { message: 'tldr must be 100 words or fewer' }),
  heroImage: heroImageSchema.optional(), // arrives in Plan 03-05
  comparison: comparisonSchema.optional(),
});
export type DraftFrontmatter = z.infer<typeof draftSchema>;
```
