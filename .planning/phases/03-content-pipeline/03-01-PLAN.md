```typescript
/** Registry stage vocabulary (D-09). `rejected` is terminal; the other five
 * advance left-to-right when the stage that produces them succeeds. */
export type RegistryStage = 'expanded' | 'filtered' | 'briefed' | 'outlined' | 'drafted' | 'rejected';
export type GeoCode = 'us' | 'uk' | 'ca' | 'au';
export const GEO_CODES: GeoCode[] = ['us', 'uk', 'ca', 'au'];

export interface GeoMetrics { volume: number; kd?: number; cpc?: number; }

export interface SerpResult { url: string; title: string; position: number; }

export interface ImageSlot { slot: string; alt: string; src: string; }

/** One registry entry = one keyword candidate (D-04 visible data trail: every
 * entry persists; rejected entries stay with a reason, never silently dropped). */
export interface RegistryEntry {
  id: string;                 // slug-safe id: /^[a-z0-9]+(?:-[a-z0-9]+){0,4}$/ trimmed to <=120 chars
  keyword: string;            // raw keyword/query string
  silo: SiloSlug;
  stage: RegistryStage;
  seedTopic?: string;         // seed topic string from seed-topics.json that expanded to this entry
  createdAt: string;          // ISO datetime
  updatedAt: string;          // ISO datetime
  /** Per-geo metrics (D-07: US/UK/CA/AU captured, single-locale en-US publish). */
  geo?: Partial<Record<GeoCode, GeoMetrics>>;
  metricsAsOf?: string;       // ISO — freshness gate for D-06 on-promotion refresh
  serpTop3?: SerpResult[];    // captured during expansion cannibalization check (D-05)
  serpAsOf?: string;
  paa?: string[];             // People-Also-Ask questions captured from SERP features
  rejectionReason?: string;   // required when stage === 'rejected' (D-04)
  brief?: { completedAt: string };
  outline?: { completedAt: string; lint: 'pass' };
  draft?: { slug: string; landedAt: string; attempts: number; lastError?: string };
  images?: { hero: ImageSlot; inArticle?: ImageSlot[]; completedAt: string };
}

/** Per-silo shard file (D-08): content/keywords/{silo}/registry.json */
export interface RegistryShard {
  version: 1;
  updatedAt: string;
  keywords: RegistryEntry[];
}
```

lib/pipeline/registry.ts (create):
```typescript
import { SILO_SLUGS, type SiloSlug } from '../silos';
import type { RegistryEntry, RegistryStage, RegistryShard } from './types';

// RegistryDir defaults to `<repo>/content/keywords` (resolved relative to this
// module). Every function takes `registryDir` override (used by --flag and tests).
export function loadShard(registryDir: string, silo: SiloSlug): RegistryShard;      // throws on parse/validation failure
export function saveShard(registryDir: string, silo: SiloSlug, shard: RegistryShard): void;  // JSON.stringify 2-space + trailing newline, atomic (tmp+rename)
export function listByStage(registryDir: string, stage: RegistryStage): RegistryEntry[];     // scans all 5 shards
export function listPromotable(registryDir: string, from: RegistryStage, to: RegistryStage, limit?: number): RegistryEntry[];
  // → entries with stage `from`, ordered oldest-updated first; `limit` default 1 (D-09 slot rule: one promotion per silo-pass per run, round-robin across silos)
export function advance(entry: RegistryEntry, to: RegistryStage): RegistryEntry;      // sets stage + updatedAt; throws if transition invalid (only +1 in chain or to 'rejected')
export function reject(entry: RegistryEntry, reason: string): RegistryEntry;          // stage='rejected' + rejectionReason required non-empty
export const STAGE_CHAIN: readonly RegistryStage[] = ['expanded','filtered','briefed','outlined','drafted'];
```

lib/pipeline/budget.ts (create):
```typescript
export interface SpendOp { ts: string; month: 'YYYY-MM'; service: 'dfs.keyword-ideas' | 'dfs.serp'; cost: number; keywords?: string[]; note?: string; }
export interface Budget { envelopeUsd: number; month: string; monthSpendUsd: number; remainingUsd: number; }

export function readBudget(registryDir: string, now?: Date): Budget;   // envelope from spend-ledger.json (default 25 if field absent)
export function appendSpend(registryDir: string, op: Omit<SpendOp, 'ts' | 'month'>, now?: Date): void;  // pushes op with ts/month (monthly rollover: old-month ops stay for the trail, sums gate only current-month)
export function assertSpendable(registryDir: string, maxAddlUsd: number, now?: Date): Budget;  // throws BudgetExhaustedError (name exported) when remainingUsd < maxAddlUsd
```
